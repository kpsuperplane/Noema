package main

import (
 "context"
 "bufio"
 "encoding/json"
 "fmt"
 "os"
 "path/filepath"
 "sync"

 "github.com/kpsuperplane/noema/internal/adapter"
 "github.com/kpsuperplane/noema/internal/store"
)

func main() {
 ctx := context.Background()
 root, err := os.OpenRoot(os.Args[1]); if err != nil { panic(err) }; defer root.Close()
 database, err := store.Open(ctx, filepath.Join(os.Args[1], "noema.sqlite3")); if err != nil { panic(err) }; defer database.Close()
 service, err := adapter.NewService(root, database); if err != nil { panic(err) }
 bindings, err := service.Bindings(); if err != nil { panic(err) }
 revoked := len(os.Args) > 3 && os.Args[3] == "revoked"
 if revoked { fmt.Println(`{"ready":true}`); _, _ = bufio.NewReader(os.Stdin).ReadString('\n') }
 start := make(chan struct{})
 results := make(chan bool, len(bindings))
 var group sync.WaitGroup
 count := 0
 for _, binding := range bindings {
  if binding.GrantID != os.Args[2] { continue }
  count++
  group.Go(func() {
   <-start
   payload, ok, err := service.Call(ctx, binding, json.RawMessage(`{"id":"concurrent"}`))
   var value struct { Name string `json:"name"` }
   if revoked { results <- err != nil && !ok } else { results <- err == nil && ok && json.Unmarshal(payload, &value) == nil && value.Name == "account-one record" }
  })
 }
 close(start); group.Wait(); close(results)
 passed := 0; for ok := range results { if ok { passed++ } }
 fmt.Printf("{\"calls\":%d,\"passed\":%d}\n", count, passed)
 if count != 2 || passed != 2 { os.Exit(1) }
}
